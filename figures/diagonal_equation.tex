%
% Three corrections to the inherited version:
%   (1) i,i -> i,j,k. A grid point is (i,j,k); "i,i" was never a valid index.
%   (2) a_{ii} -> d_{ijk}, so it matches the symbol used in Eq.(loss).
%   (3) the Robin term is added -- the code includes it in the diagonal:
%           diag_coeff = k*V_m + sum(six face conductances) + coeff_integral_au
% ---------------------------------------------------------------

In this work we use the Jacobi preconditioner: every element of the left-hand side
$(A u)$ and right-hand side $(b)$ is divided by the diagonal coefficient of its own
row. For the cell centred on $\mathbf{x}_{ijk}$ this is the quantity $d_{ijk}$
appearing in~\eqref{eq:loss},
\begin{equation}
\begin{aligned}
  d_{ijk} = \; & k_{ijk}\,\bigl|V^{-}_{ijk}\bigr|
  \;+\; \frac{\mu_{i-\frac12,j,k} A^{-}_{i-\frac12,j,k}
            + \mu_{i+\frac12,j,k} A^{-}_{i+\frac12,j,k}}{\Delta x}
  \;+\; \frac{\mu_{i,j-\frac12,k} A^{-}_{i,j-\frac12,k}
            + \mu_{i,j+\frac12,k} A^{-}_{i,j+\frac12,k}}{\Delta y} \\[4pt]
  & +\; \frac{\mu_{i,j,k-\frac12} A^{-}_{i,j,k-\frac12}
            + \mu_{i,j,k+\frac12} A^{-}_{i,j,k+\frac12}}{\Delta z}
  \;+\; \frac{\mu_{\Gamma}\displaystyle\int_{\Gamma \cap V_{ijk}} \alpha \, dA}
             {\mu_{\Gamma} + \alpha_{\Gamma}\,\bigl|\delta_{ijk}\bigr|} ,
\end{aligned}
\label{eq:diagonal}
\end{equation}
in which $A^{-}$ denotes the portion of a cell face lying in $\Omega^{-}$,
$\bigl|V^{-}_{ijk}\bigr|$ the corresponding partial cell volume, and $\mu_{\Gamma}$,
$\alpha_{\Gamma}$ the coefficients evaluated at the projection of $\mathbf{x}_{ijk}$
onto $\Gamma$. The final term is the diagonal contribution of the Robin condition,
arising from the same Taylor projection as~\eqref{eq:taylor_robin}; to leading order in
$\delta_{ijk}$ it reduces to $\int_{\Gamma \cap V_{ijk}} \alpha \, dA$, and it vanishes
for cells the interface does not cross. Note that for memory efficiency the matrix is
never explicitly assembled; only the action $A u$ is computed.
